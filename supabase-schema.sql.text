-- ---------- helper: auto updated_at ----------
create or replace function public.set_updated_at()
returns trigger language plpgsql as $$
begin
  new.updated_at = now();
  return new;
end $$;

-- ---------- profiles (app users: doctor + apprentice) ----------
create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  email text not null,
  name text not null,
  role text not null default 'apprentice' check (role in ('doctor','apprentice')),
  created_at timestamptz not null default now()
);

-- ---------- patients ----------
create table if not exists public.patients (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  age int,
  sex text check (sex in ('male','female','other')),
  phone text,
  weight_kg numeric,
  allergies text default '',
  chronic text default '',
  current_meds text default '',
  pregnant boolean default false,
  renal_issue boolean default false,
  notes text default '',
  is_deleted boolean not null default false,
  created_by uuid references public.profiles(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
create index if not exists idx_patients_name on public.patients (name);
create index if not exists idx_patients_updated on public.patients (updated_at);
drop trigger if exists trg_patients_updated on public.patients;
create trigger trg_patients_updated before update on public.patients
  for each row execute function public.set_updated_at();

-- ---------- visits ----------
create table if not exists public.visits (
  id uuid primary key default gen_random_uuid(),
  patient_id uuid not null references public.patients(id) on delete cascade,
  visit_date date not null default current_date,
  chief_complaint text default '',
  history text default '',
  examination text default '',
  diagnosis text default '',
  plan text default '',
  next_visit_date date,
  fee_consult numeric not null default 0,
  fee_medicine numeric not null default 0,
  created_by uuid references public.profiles(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
create index if not exists idx_visits_patient on public.visits (patient_id);
create index if not exists idx_visits_date on public.visits (visit_date);
create index if not exists idx_visits_updated on public.visits (updated_at);
drop trigger if exists trg_visits_updated on public.visits;
create trigger trg_visits_updated before update on public.visits
  for each row execute function public.set_updated_at();

-- ---------- visit_templates (SmartPhrase-style) ----------
create table if not exists public.visit_templates (
  id uuid primary key default gen_random_uuid(),
  code text not null unique,
  title text not null,
  field text not null default 'examination'
    check (field in ('chief_complaint','history','examination','plan')),
  body text not null default '',
  created_by uuid references public.profiles(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
drop trigger if exists trg_templates_updated on public.visit_templates;
create trigger trg_templates_updated before update on public.visit_templates
  for each row execute function public.set_updated_at();

-- ---------- drugs ----------
create table if not exists public.drugs (
  id uuid primary key default gen_random_uuid(),
  generic text not null unique,
  brands text default '',
  formulations text default '',
  category text default '',
  indications text default '',
  dose_adult text default '',
  dose_ped text default '',
  ped_flag text default 'no_data' check (ped_flag in ('ok','caution','contra','no_data')),
  ped_mgkg numeric,
  ped_mgkg_max numeric,
  ped_daymax_mgkg numeric,
  ped_cap_mg numeric,
  ped_freq text,
  dose_renal text default '',
  ci text default '',
  contra text default '',
  precautions text default '',
  contra_keys jsonb not null default '[]',
  interactions jsonb not null default '[]',
  allergy text default '',
  preg text default '',
  mon text default '',
  custom boolean not null default false,
  updated_at timestamptz not null default now()
);
create index if not exists idx_drugs_generic on public.drugs (generic);
create index if not exists idx_drugs_cat on public.drugs (category);
drop trigger if exists trg_drugs_updated on public.drugs;
create trigger trg_drugs_updated before update on public.drugs
  for each row execute function public.set_updated_at();

-- ---------- prescriptions ----------
create table if not exists public.prescriptions (
  id uuid primary key default gen_random_uuid(),
  visit_id uuid references public.visits(id) on delete set null,
  patient_id uuid not null references public.patients(id) on delete cascade,
  presc_date date not null default current_date,
  notes text default '',
  warnings_overridden jsonb not null default '[]',
  created_by uuid references public.profiles(id),
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()

create index if not exists idx_presc_patient on public.prescriptions (patient_id);
drop trigger if exists trg_presc_updated on public.prescriptions;
create trigger trg_presc_updated before update on public.prescriptions
  for each row execute function public.set_updated_at();

-- ---------- prescription_items ----------
create table if not exists public.prescription_items (
  id uuid primary key default gen_random_uuid(),
  prescription_id uuid not null references public.prescriptions(id) on delete cascade,
  drug_generic text not null,
  formulation text default '',
  dose_mg numeric,
  dose_unit text default 'mg',
  frequency text default '',
  duration_days int,

  unit_price numeric not null default 0,

  created_at timestamptz not null default now()
);

-- ---------- dispensing_log (sales analysis source) ----------
create table if not exists public.dispensing_log (
  id uuid primary key default gen_random_uuid(),
  prescription_item_id uuid references public.prescription_items(id) on delete set null,
  patient_id uuid references public.patients(id) on delete set null,
  drug_generic text not null,
  qty numeric not null default 0,
  unit_price numeric not null default 0,
  total numeric not null default 0,

  created_at timestamptz not null default now()
);
create index if not exists idx_disp_date on public.dispensing_log (dispensed_date);
create index if not exists idx_disp_drug on public.dispensing_log (drug_generic);

-- ---------- followups ----------

-- helper: is the current user the doctor?
create or replace function public.is_doctor()
returns boolean language sql stable as $$
  select exists (
    select 1 from public.profiles
    where id = auth.uid() and role = 'doctor'
  );
$$;

-- First registered user becomes the doctor (called once by the app after signup).
-- Secure: only succeeds when no doctor exists yet.
create or replace function public.claim_first_doctor()
returns boolean language plpgsql security definer as $$
begin
  if exists (select 1 from public.profiles where role = 'doctor') then
    return false;
  end if;
  update public.profiles set role = 'doctor' where id = auth.uid();
  return true;
end $$;
grant execute on function public.claim_first_doctor() to authenticated;

-- profiles: everyone authenticated can read; new users insert own row as apprentice only;
-- role can only be raised by an existing doctor (prevents privilege escalation).
-- BOOTSTRAP: after the very first signup, run in SQL Editor:
--   update public.profiles set role='doctor' where email='YOUR_EMAIL';
drop policy if exists p_profiles_select on public.profiles;
create policy p_profiles_select on public.profiles for select to authenticated using (true);
drop policy if exists p_profiles_insert on public.profiles;
create policy p_profiles_insert on public.profiles for insert to authenticated
  with check (id = auth.uid() and role = 'apprentice');
drop policy if exists p_profiles_update on public.profiles;
create policy p_profiles_update on public.profiles for update to authenticated
  using (id = auth.uid() or public.is_doctor())
  with check (public.is_doctor() or (id = auth.uid() and role = 'apprentice'));
drop policy if exists p_profiles_delete on public.profiles;
create policy p_profiles_delete on public.profiles for delete to authenticated using (public.is_doctor());

-- generic clinic tables: authenticated read/write, delete = doctor only
do $$
declare t text;
begin
  foreach t in array array['patients','visits','visit_templates','prescriptions','prescription_items','dispensing_log','followups']
  loop
    execute format('drop policy if exists p_%s_select on public.%s', t, t);
    execute format('create policy p_%s_select on public.%s for select to authenticated using (true)', t, t);
    execute format('drop policy if exists p_%s_insert on public.%s', t, t);
    execute format('create policy p_%s_insert on public.%s for insert to authenticated with check (true)', t, t);
    execute format('drop policy if exists p_%s_update on public.%s', t, t);
    execute format('create policy p_%s_update on public.%s for update to authenticated using (true) with check (true)', t, t);
    execute format('drop policy if exists p_%s_delete on public.%s', t, t);
    execute format('create policy p_%s_delete on public.%s for delete to authenticated using (public.is_doctor())', t, t);
  end loop;
end $$;

-- drugs: authenticated read/insert/update; delete = doctor only
drop policy if exists p_drugs_select on public.drugs;
create policy p_drugs_select on public.drugs for select to authenticated using (true);
drop policy if exists p_drugs_insert on public.drugs;
create policy p_drugs_insert on public.drugs for insert to authenticated with check (true);
drop policy if exists p_drugs_update on public.drugs;
create policy p_drugs_update on public.drugs for update to authenticated using (true) with check (true);
drop policy if exists p_drugs_delete on public.drugs;
create policy p_drugs_delete on public.drugs for delete to authenticated using (public.is_doctor());
